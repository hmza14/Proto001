CREATE TABLE IF NOT EXISTS public.usr_dat
(
    usr_keyu integer NOT NULL,
    usr_actf boolean DEFAULT true,
    usr_name character varying(50) COLLATE pg_catalog."default" NOT NULL,
    usr_mtrc character varying(50) COLLATE pg_catalog."default" NOT NULL,
    usr_type character varying(50) COLLATE pg_catalog."default" NOT NULL,
    usr_lang character varying(50) COLLATE pg_catalog."default" NOT NULL,
    usr_limi date NOT NULL,
    usr_mail character varying(128) COLLATE pg_catalog."default" NOT NULL,
    usr_logn character varying(20) COLLATE pg_catalog."default" NOT NULL,
    usr_pass character varying(100) COLLATE pg_catalog."default" NOT NULL,
    usr_stat character varying(3) COLLATE pg_catalog."default" DEFAULT '020'::character varying,
    usr_crda date,
    usr_crhe character varying(6) COLLATE pg_catalog."default",
    usr_crqi character varying(20) COLLATE pg_catalog."default",
    usr_moda date,
    usr_mohe character varying(6) COLLATE pg_catalog."default",
    usr_moqi character varying(20) COLLATE pg_catalog."default",
    CONSTRAINT usr_dat_pkey PRIMARY KEY (usr_keyu),
    CONSTRAINT usr_dat_usr_logn_unique UNIQUE (usr_logn),
    CONSTRAINT usr_dat_usr_type_check CHECK (usr_type IN ('Type1', 'Type2', 'Type3'))
)

    OWNER to postgres;
